const express = require('express'); const mongoose = require('mongoose'); const cors = require('cors'); const app = express(); app.use(express.json()); app.use(cors()); // ========================================== // 1. MONGODB DATABASE MODEL // ========================================== mongoose.connect('mongodb://localhost:27017/social_exchange', { useNewUrlParser: true, useUnifiedTopology: true }); const userSchema = new mongoose.Schema({ username: { type: String, required: true }, coins: { type: Number, default: 0 }, likedVideosCount: { type: Number, default: 0 }, followedAccountsCount: { type: Number, default: 0 } }); const User = mongoose.model('User', userSchema); // ========================================== // 2. BACKEND API ENDPOINTS // ========================================== // डमी या नया यूजर पाने/बनाने का API app.get('/api/get-user', async (req, res) => { try { let user = await User.findOne(); if (!user) { user = await User.create({ username: "demo_user", coins: 0 }); } res.json({ success: true, user }); } catch (err) { res.status(500).json({ success: false, error: err.message }); } }); // टॉस्क पूरा करने पर कॉइन देने का लॉजिक (5 लाइक + 2 फॉलो = 1 कॉइन) app.post('/api/complete-task', async (req, res) => { const { userId, actionType } = req.body; try { const user = await User.findById(userId); if (!user) return res.status(404).json({ success: false, message: "यूजर नहीं मिला" }); if (actionType === 'like') { user.likedVideosCount += 1; } else if (actionType === 'follow') { user.followedAccountsCount += 1; } // रिवॉर्ड नियम: 5 लाइक और 2 फॉलो पर 1 कॉइन if (user.likedVideosCount >= 5 && user.followedAccountsCount >= 2) { user.coins += 1; user.likedVideosCount -= 5; user.followedAccountsCount -= 2; } await user.save(); res.json({ success: true, coins: user.coins, likedVideosCount: user.likedVideosCount, followedAccountsCount: user.followedAccountsCount }); } catch (err) { res.status(500).json({ success: false, error: err.message }); } }); // कॉइन से सर्विस खरीदने का लॉजिक (1 कॉइन = 100 फॉलोअर / 100 सब्सक्राइबर / 20 लाइक्स) app.post('/api/buy-service', async (req, res) => { const { userId, serviceType, coinsToUse, targetUrl } = req.body; try { const user = await User.findById(userId); if (!user) return res.status(404).json({ success: false, message: "यूजर नहीं मिला" }); if (user.coins < coinsToUse) { return res.status(400).json({ success: false, message: "पर्याप्त कॉइन नहीं हैं!" }); } let itemQuantity = 0; if (serviceType === 'followers') { itemQuantity = coinsToUse * 100; // 1 Coin = 100 Followers } else if (serviceType === 'subscribers') { itemQuantity = coinsToUse * 100; // 1 Coin = 100 Subscribers } else if (serviceType === 'likes') { itemQuantity = coinsToUse * 20; // 1 Coin = 20 Likes } else { return res.status(400).json({ success: false, message: "गलत सर्विस का चयन किया गया" }); } user.coins -= coinsToUse; await user.save(); res.json({ success: true, message: `सफलतापूर्वक ${itemQuantity} ${serviceType} का ऑर्डर दिया गया!`, remainingCoins: user.coins }); } catch (err) { res.status(500).json({ success: false, error: err.message }); } }); // ========================================== // 3. FRONTEND UI ROUTE (SINGLE FILE INTEGRATION) // ========================================== app.get('/', (req, res) => { res.send(` सोशल कॉइन एक्सचेंज प्लेटफॉर्म

`); }); // ========================================== // 4. SERVER LISTEN // ========================================== const PORT = 5000; app.listen(PORT, () => { console.log(`सर्वर चालू है: http://localhost:${PORT}`); });
